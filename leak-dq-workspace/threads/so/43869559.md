## Question

```
  void assign(const Matrix& src, Matrix& dest) {
  int nr = src.rows();
  int nc = src.cols();
#pragma omp parallel for num_threads(NUM_THREADS) firstprivate(nr, nc) schedule(dynamic, 1)
  for (int32 c = 0 ; c < nc ; ++c) {
    for (int32 r = 0 ;  r < nr ; ++r)
      mat(r,c) = src(r,c);
  }

```
