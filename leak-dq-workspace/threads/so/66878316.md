EC_KEY *ec_from_pub(uint8_t const pub[EC_PUB_LEN]){

  EC_KEY *key = NULL;
  EC_POINT *pubKey = NULL;
  const EC_GROUP *group = NULL;
  BN_CTX *ctx;
  char *buf;
  int i;

  buf = malloc(sizeof(char)*(EC_PUB_LEN + 1));

  for(i=0;i<EC_PUB_LEN;i++){
    buf[i] = (char) pub[i];
  }

  key = EC_KEY_new_by_curve_name(EC_CURVE);

  if(key == NULL)
    return NULL;

  group = EC_KEY_get0_group(key);

  if(group == NULL){
    EC_KEY_free(key);
    return NULL;
  }

  pubKey = EC_POINT_new(group);
  ctx = BN_CTX_new();
  EC_POINT_hex2point(group, buf, pubKey, ctx);

  if(pubKey == NULL){
    EC_KEY_free(key);
    BN_CTX_free(ctx);
    return NULL;
  }

  EC_KEY_set_public_key(key, pubKey);
  EC_POINT_free(pubKey);
  BN_CTX_free(ctx);
  free(buf);
  return key;
}

  BN_CTX *ctx;

    return NULL;

## Comment on question 66878316 118220620

responder · dave_thompson_085 · 2021-03-30T22:24:45Z · score 1 · https://stackoverflow.com/questions/66878316/#comment118220620_66878316

You are trying to put a secp256k1 uncompressed point (publickey) as hex in a 65-byte buffer; that is wrong. The *octetstring* format of such a point is 65 *bytes*, but the hex format is 131 *characters* (65 times 2, plus a null terminator). Also the pointer returned from `EC_POINT_point2hex` is allocated by `OPENSSL_malloc` and should be freed by `OPENSSL_free` not plain `free`. `EC_KEY_check_key` returns type `int` with 1 for true and 0 for false (effectively boolean, but C before C99 doesn't have boolean), but cannot be null, because only characters and pointers can be null.
