My actual programme crashed due to increasing memory in "RES" section from "top" command in ubuntu linux which is why I wanted to track the memory leak. Here "RES" refers to the physical memory being used by the program at run time. So, I wrote a little programme to keep track of "RES" memory caused by malloc() and free() operation :

```
#include <stdio.h>
#include <stdlib.h>
#include <sys/time.h>
#include <unistd.h>
#include <sys/resource.h>

#define CAPACITY_DEBUG 1000000

void printMemUsage(){
    struct rusage usage;
    getrusage(RUSAGE_SELF, &usage);
    fprintf(stderr,"Memory usage: %ld KB .................\n", usage.ru_maxrss);
}

void MallocTest(){
    fprintf(stderr,"test 1 started ......................!!\n");
    printMemUsage();
    sleep(5);

    fprintf(stderr,"starting to allocate ....................\n");
    int* arr = (int*)malloc(CAPACITY_DEBUG * sizeof(int));
    if (arr == NULL) {
        fprintf(stderr, "Memory allocation failed. Exiting...\n");
        return;
    }
    for(int i = 0; i < CAPACITY_DEBUG; i++){
        arr[i] = 5;
    }
    printMemUsage();
    fprintf(stderr,"allocation done, going to sleep................\n");
    sleep(15);

    fprintf(stderr,"Woke up, starting to free............\n");
    free(arr); // Free the memory when you're done.
    // malloc_trim(0);
    fprintf(stderr,"free done................\n");
    sleep(5);
    printMemUsage();
}

void MallocTest2(){
    fprintf(stderr,"test 2 started ......................!!\n");
    printMemUsage();
    sleep(5);

    fprintf(stderr,"starting to allocate ....................\n");
    int* arr[CAPACITY_DEBUG];  // Declare an array of integer pointers
    for (int i = 0; i < CAPACITY_DEBUG; i++) {
        arr[i] = (int*)malloc(sizeof(int));  // Allocate memory for each element
        if (arr[i] == NULL) {
            // Handle memory allocation failure
            perror("malloc");
            exit(1);
        }
        // Initialize the value in the allocated memory
        *arr[i] = i * 2;
    }
    printMemUsage();
    fprintf(stderr,"allocation done, going to sleep................\n");
    sleep(15);

    fprintf(stderr,"Woke up, starting to free............\n");
    // Don't forget to free the allocated memory when done
    for (int i = 0; i < CAPACITY_DEBUG; i++) {
        free(arr[i]);
    }
    //malloc_trim(0);
    fprintf(stderr,"free done................\n");
    sleep(30);
    printMemUsage();
}

int main(){

    MallocTest2();
    
    return 1;
}

```

`int* arr[CAPACITY_DEBUG];` would try to burn up 8Mb of stack space, which is probably more than your compiler is setup to provide for the executable... You may have to use a smaller number than 1 million pointers in this array...
